def _signal_group(pgid: int, signum: int) -> bool:
    """Signal every process in `pgid`. False when the group is already gone."""
    if pgid <= 0:
        return False
    try:
        os.killpg(pgid, signum)
    except (ProcessLookupError, PermissionError):
        return False
    return True
