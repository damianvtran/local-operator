"""Capture the settings page AT THE BOTTOM of the list, reached by the WHEEL.

The state the bounce report is about. `settings_shot.py`'s existing states all
place the cursor with a key and then scroll to it, so none of them can show what
a wheel gesture does to the viewport -- which is the whole subject here.

    env -u NO_COLOR TERM=xterm-256color .venv/bin/python \
        /tmp/scroll_shot.py OUT.svg COLSxROWS WORKTREE
"""
import asyncio, os, sys, tempfile
from pathlib import Path
OUT = sys.argv[1]
COLS, ROWS = (int(p) for p in sys.argv[2].split("x"))
WT = sys.argv[3]
sys.path.insert(0, WT)
_S = tempfile.mkdtemp(prefix="lo-scrollshot-")
os.environ["LOCAL_OPERATOR_CONFIG_DIR"] = _S
from textual import events  # noqa: E402
from local_operator.config import ConfigManager  # noqa: E402
from local_operator.tui.app import OperatorApp  # noqa: E402
from local_operator.tui.widgets.settings_view import SettingsView  # noqa: E402
from local_operator.tui.widgets.assistant import AssistantBlock  # noqa: E402
from local_operator.tui.widgets.transcript import UserBlock  # noqa: E402
from tests.unit.tui.test_app_pilot import FakeSession, _factory  # noqa: E402

TEAMS = [("lopdev", "manager · 6 roles", "ships local-operator changes end to end")]
AGENTS = [("coder", "role · effort med", "implements one bounded slice")]
PROVIDERS = [("anthropic", "signed in"), ("openrouter", "api key")]

def wheel(w, down=True):
    k = events.MouseScrollDown if down else events.MouseScrollUp
    return k(widget=w, x=1, y=1, delta_x=0, delta_y=1 if down else -1,
             button=0, shift=False, meta=False, ctrl=False)

def seed():
    m = ConfigManager(Path(_S))
    m.set_config_value("hosting", "anthropic")
    m.set_config_value("model_name", "claude-opus-5")
    m.set_config_value("tool_approval_mode", "auto")

async def main():
    seed()
    app = OperatorApp(lambda: _factory(FakeSession()))
    async with app.run_test(size=(COLS, ROWS)) as pilot:
        await pilot.pause()
        for t in range(1, 3):
            app._append_block(UserBlock(f"Turn {t}: can I change the default model?"))
            b = AssistantBlock(); b.update_text(f"Answer {t}: yes, /settings has it."); app._append_block(b)
        await pilot.pause()
        app._open_settings_view()
        v = app.query_one(SettingsView)
        v.load(teams=TEAMS, agents=AGENTS, providers=PROVIDERS)
        await pilot.pause()
        body = v._body
        # THE GESTURE: wheel to the bottom and keep going, which is where the
        # bounce fired. 40 notches is comfortably past max_scroll_y at every
        # size captured here.
        for _ in range(40):
            v._list.post_message(wheel(v._list))
            await pilot.pause()
        app.save_screenshot(OUT)
        off = body.scroll_offset.y
        print(f"size=({COLS}, {ROWS}) scroll_y={off}/{body.max_scroll_y} "
              f"at_bottom={off == body.max_scroll_y} selected={v._selected} "
              f"cursor_on_screen={off <= v._selected < off + body.size.height} "
              f"rows={len(v._rows)}")
asyncio.run(main())
