"""Capture the U1 gesture: cursor parked on a BOOL, wheeled off screen, `enter`.

The state UX round 1 U1 is about. `scroll_shot.py` captures the wheel reaching
the bottom of the list; it cannot show this, because the subject here is what a
WRITING key does once the wheel has left the cursor behind. Two frames per tree,
before and after the press, because the finding is precisely that they were
identical while config.yml changed.

    env -u NO_COLOR TERM=xterm-256color .venv/bin/python \
        /tmp/probe447/reveal_shot.py OUT-PREFIX COLSxROWS WORKTREE
"""

import asyncio
import os
import sys
import tempfile
from pathlib import Path

PREFIX = sys.argv[1]
COLS, ROWS = (int(p) for p in sys.argv[2].split("x"))
WT = sys.argv[3]
sys.path.insert(0, WT)
_S = tempfile.mkdtemp(prefix="lo-revealshot-")
os.environ["LOCAL_OPERATOR_CONFIG_DIR"] = _S

from textual import events  # noqa: E402

from local_operator.config import ConfigManager  # noqa: E402
from local_operator.tui.app import OperatorApp  # noqa: E402
from local_operator.tui.widgets.assistant import AssistantBlock  # noqa: E402
from local_operator.tui.widgets.settings_view import SettingsView  # noqa: E402
from local_operator.tui.widgets.transcript import UserBlock  # noqa: E402
from tests.unit.tui.test_app_pilot import FakeSession, _factory  # noqa: E402

TEAMS = [("lopdev", "manager · 6 roles", "ships local-operator changes end to end")]
AGENTS = [("coder", "role · effort med", "implements one bounded slice")]
PROVIDERS = [("anthropic", "signed in"), ("openrouter", "api key")]


def wheel(w, down=True):
    k = events.MouseScrollDown if down else events.MouseScrollUp
    return k(
        widget=w, x=1, y=1, delta_x=0, delta_y=1 if down else -1,
        button=0, shift=False, meta=False, ctrl=False,
    )


def seed():
    m = ConfigManager(Path(_S))
    m.set_config_value("hosting", "anthropic")
    m.set_config_value("model_name", "claude-opus-5")
    m.set_config_value("tool_approval_mode", "auto")


def cfg_bytes() -> bytes:
    p = os.path.join(_S, "config.yml")
    return open(p, "rb").read() if os.path.exists(p) else b""


async def main():
    seed()
    app = OperatorApp(lambda: _factory(FakeSession()))
    async with app.run_test(size=(COLS, ROWS)) as pilot:
        await pilot.pause()
        for t in range(1, 3):
            app._append_block(UserBlock(f"Turn {t}: can I change the default model?"))
            b = AssistantBlock()
            b.update_text(f"Answer {t}: yes, /settings has it.")
            app._append_block(b)
        await pilot.pause()
        app._open_settings_view()
        v = app.query_one(SettingsView)
        v.load(teams=TEAMS, agents=AGENTS, providers=PROVIDERS)
        await pilot.pause()
        body = v._body

        # Park the cursor on a BOOL the user chose deliberately, then wheel
        # away from it — the gesture that makes the cursor invisible.
        for i, row in enumerate(v._rows):
            if row.kind == "setting" and row.setting and row.setting.key == "retry.enabled":
                v._selected = i
                break
        v._repaint()
        v._scroll_to_selection()
        await pilot.pause()
        for _ in range(15):
            v._list.post_message(wheel(v._list))
            await pilot.pause()

        off = body.scroll_offset.y
        h = body.size.height
        before_bytes = cfg_bytes()
        app.save_screenshot(f"{PREFIX}-pre-enter.svg")
        print(
            f"size=({COLS}, {ROWS}) PRE-ENTER  scroll_y={off}/{body.max_scroll_y} "
            f"selected={v._selected} cursor_on_screen={off <= v._selected < off + h}"
        )

        await pilot.press("enter")
        await pilot.pause()
        off2 = body.scroll_offset.y
        app.save_screenshot(f"{PREFIX}-post-enter.svg")
        print(
            f"size=({COLS}, {ROWS}) POST-ENTER scroll_y={off2}/{body.max_scroll_y} "
            f"selected={v._selected} cursor_on_screen={off2 <= v._selected < off2 + h} "
            f"config_changed={cfg_bytes() != before_bytes} viewport_moved={off2 != off}"
        )


asyncio.run(main())
