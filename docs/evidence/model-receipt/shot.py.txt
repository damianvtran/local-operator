"""Frame the `/model` receipt on a session whose label updates ASYNCHRONOUSLY.

Mirrors `RemoteSession.set_model` (an `asyncio.create_task` against the owner;
`model_label` reads the frontend-state sync that lands later). Usage:

    env -u NO_COLOR TERM=xterm-256color .venv/bin/python /tmp/lop-shot-model-receipt.py out.svg
"""

import asyncio
import os
import sys

sys.path.insert(0, os.getcwd())  # repo root, so `tests.` imports resolve

from local_operator.tui.app import OperatorApp  # noqa: E402
from tests.unit.tui.test_app_pilot import (  # noqa: E402
    FakeSession,
    _AccessController,
    _await_session,
    _factory,
    _transcript_text,
)


class _AsyncLabelSession(FakeSession):
    def __init__(self) -> None:
        super().__init__()
        self._label = "anthropic/claude-opus-5"

    @property
    def model_label(self) -> str:
        return self._label

    def set_model(self, model, *, explicit: bool = False) -> None:
        async def _land() -> None:
            await asyncio.sleep(0.5)
            self._label = f"{model.provider}/{model.model_id}"

        asyncio.get_running_loop().create_task(_land())


async def main() -> None:
    session = _AsyncLabelSession()
    ctrl = _AccessController(stored=("openrouter", "anthropic"))
    app = OperatorApp(lambda: _factory(session), provider_controller=ctrl)
    async with app.run_test(size=(110, 24)) as pilot:
        await _await_session(app, pilot)
        app._run_slash_command("/model anthropic/claude-fable-5-1")
        await pilot.pause()
        await pilot.pause()
        app.save_screenshot(sys.argv[1])
        print(_transcript_text(app))


asyncio.run(main())
