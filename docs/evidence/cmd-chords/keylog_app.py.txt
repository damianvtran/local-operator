"""Run the REAL OperatorApp in a real terminal, logging every key the
Editor receives plus the resulting composer/clipboard state.

This is not a synthesised event: the app is the production one, the driver is
Textual's real linux_driver on a real pty, and the keys are whatever the
terminal actually delivers.
"""
import os, sys, datetime
sys.path.insert(0, "/tmp/lop-cmd")

LOG = os.environ.get("LO_KEYLOG", "/tmp/kprobe2/keys.log")

def log(msg):
    with open(LOG, "a") as f:
        f.write("%s %s\n" % (datetime.datetime.now().strftime("%H:%M:%S.%f"), msg))
        f.flush()

from local_operator.tui.widgets.editor import Editor
from local_operator.tui import app as app_mod

_orig_on_key = Editor._on_key
async def _logged_on_key(self, event):
    log("KEY key=%r char=%r text=%r sel=%r" % (
        event.key, event.character, self.text[:60], tuple(self.selection)))
    return await _orig_on_key(self, event)
Editor._on_key = _logged_on_key

_orig_paste = Editor.action_system_paste
async def _logged_paste(self):
    log("ACTION system_paste ENTER text=%r" % self.text[:60])
    r = await _orig_paste(self)
    log("ACTION system_paste EXIT  text=%r" % self.text[:60])
    return r
Editor.action_system_paste = _logged_paste

_orig_copy = Editor.action_copy
def _logged_copy(self):
    log("ACTION copy ENTER sel=%r selected=%r" % (tuple(self.selection), self.selected_text[:40]))
    try:
        r = _orig_copy(self)
        log("ACTION copy OK")
        return r
    except Exception as e:
        log("ACTION copy RAISED %s" % type(e).__name__)
        raise
Editor.action_copy = _logged_copy

_orig_on_paste = Editor._on_paste
async def _logged_on_paste(self, event):
    log("PASTE EVENT text=%r" % (event.text[:60],))
    return await _orig_on_paste(self, event)
Editor._on_paste = _logged_on_paste

if hasattr(app_mod.OperatorApp, "on_editor_copied"):
    _oc = app_mod.OperatorApp.on_editor_copied
    def _logged_oc(self, message):
        log("APP on_editor_copied text=%r" % (message.text[:60],))
        return _oc(self, message)
    app_mod.OperatorApp.on_editor_copied = _logged_oc

_orig_click = Editor._on_click
async def _logged_click(self, event):
    log("CLICK chain=%s off=%r sel_before=%r" % (getattr(event,'chain',None), (event.x,event.y), tuple(self.selection)))
    r = await _orig_click(self, event)
    log("CLICK done sel_after=%r click_sel=%r" % (tuple(self.selection), getattr(self,'_click_selection',None)))
    return r
Editor._on_click = _logged_click

log("=== BOOT argv=%r TERM=%r" % (sys.argv, os.environ.get("TERM")))
from local_operator.cli import main
sys.argv = ["lop"]
main()
