"""Render the subagent page across the launch race that latched the footer.

The frame this captures is the operator's report: a child page whose durable
``transcript.jsonl`` did not exist when the page opened (``comms.attach`` binds
``session_dir`` before the first append lands), and which then gains it a moment
later while the page stays open. Everything below drives the REAL ``OperatorApp``
so ``local_operator.tcss`` is applied and the footer is the one a user reads.

    env -u NO_COLOR TERM=xterm-256color .venv/bin/python /tmp/histshot/shot.py \
        <repo-root> <out.svg>

No clock is patched and no flag is poked: the refreshes below are the same
``show()`` the 1 Hz job poll issues, so whatever the footer says here is what it
says in the product.
"""

from __future__ import annotations

import asyncio

import sys
import tempfile
from pathlib import Path

ROOT = sys.argv[1]
OUT = sys.argv[2]
sys.path.insert(0, ROOT)

from local_operator.session.transcript import Transcript  # noqa: E402
from local_operator.harness.types import Message  # noqa: E402
from local_operator.tui.app import OperatorApp  # noqa: E402
from local_operator.tui.widgets.subagent_view import SubagentView  # noqa: E402

from tests.unit.tui.test_band_panels import FakeSession, _async_factory, _fake_jobs  # noqa: E402
from tests.unit.tui.test_subagent_view import TRAJECTORY, _job_with, _open, _wait_history  # noqa: E402


async def main() -> None:
    tmp = Path(tempfile.mkdtemp())
    child_dir = tmp / "child"
    # attach() made the directory; the child's first append has not landed.
    child_dir.mkdir()

    job = _job_with(TRAJECTORY, status="running")
    session = FakeSession()
    session.jobs = _fake_jobs(job)
    session._subagent_comms = type(
        "Comms", (), {"session_dir_of": lambda self, _job_id: child_dir}
    )()

    app = OperatorApp(_async_factory(session))
    async with app.run_test(size=(100, 30)) as pilot:
        view = await _open(pilot, app, job)
        await _wait_history(pilot, view)
        print("after initial read: footer =", repr(view._history_state_text()))

        # The child writes its first durable rows, exactly as a live child
        # does — through the real Transcript, so the row on disk is the row
        # the product writes rather than a hand-rolled approximation of it.
        transcript = Transcript(child_dir)
        await transcript.append_message(
            Message.assistant("Durable row from the child transcript.", id="durable-1")
        )

        # The 1 Hz job poll, at its real cadence and against the real clock.
        for _ in range(10):
            await asyncio.sleep(0.2)
            app._refresh_subagent_view(view.job_id)
            await pilot.pause()
        await _wait_history(pilot, view)
        await pilot.pause()

        print("after the transcript appeared: footer =", repr(view._history_state_text()))
        print("durable rows loaded:", len(view._history_entries))
        print("flags:", {
            k: v for k, v in vars(view).items()
            if k.startswith("_history_") and isinstance(v, (bool, str, type(None)))
        })
        app.save_screenshot(OUT)


asyncio.run(main())
