def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def relative(path):
    return str(path.relative_to(ROOT))

def put(path, text):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('x', encoding='utf-8') as f:
        f.write(text)

def dump(path, obj):
    put(path, json.dumps(obj, ensure_ascii=False, indent=2) + '\n')

def load(path):
    def no_duplicates(pairs):
        result = {}
        for k, v in pairs:
            assert k not in result, ('duplicate JSON key', path, k)
            result[k] = v
        return result
    return json.loads(path.read_text(encoding='utf-8'), object_pairs_hook=no_duplicates)

                'These are offline catalogue renders, not screenshots or full readiness integration.', '']
    categories_seen = set()
    render_count = 0
    for key, english in source.items():
        text = target[key]
        assert isinstance(text, str) and text
        assert not (set(text) & CONTROLS), (locale, key, 'bidi controls')
        assert not any(unicodedata.category(c) == 'Cc' for c in text), (locale,key,'control')
        if locale == 'vi':

        assert runtime.message_arguments(english) == runtime.message_arguments(text), (locale, key, 'arguments/order')
        plural = runtime.plural_selectors(text)
        for selectors in plural:
            assert set(selectors) == CATEGORIES[locale], (locale, key, selectors)
        for argument in runtime.plural_arguments(text):
            assert not any(b.empty for b in argument.branches)
        expected_literals = set(re.findall(r'`([^`]+)`', english))
        assert expected_literals == set(re.findall(r'`([^`]+)`', text)), (locale,key,'commands')
        # Audit loss of a parameter or literal in ONE branch, not just a union
        # across all branches. All plural keys in this namespace are whole-message.
        for node in runtime.parse_message(text):
            if getattr(node, 'kind', None) == 'plural':
                expected_params = {n for n,k in runtime.message_arguments(english) if k != 'plural'}
                for selector, body in node.parts:
                    names = {n.name for n in body if hasattr(n, 'name')}
                    assert names == expected_params, (locale,key,selector,'branch params')
                    branch_literals = set()
                    for n in body:
                        if hasattr(n, 'value'):
                            branch_literals.update(re.findall(r'`([^`]+)`', n.value))
                    assert branch_literals == expected_literals, (locale,key,selector,'branch commands')
        log.append(f'PASS {key}: key/ICU/argument-order/branch-values/commands/Unicode; selectors={plural!r}')
        for count in (COUNTS if plural else [1]):
            params = {'count': count, 'device': 'host-sd2', 'pids': '4101, 4102',
                      'holders': 'worker.service', 'builds': 'v0.99.1+running',
                      'recorded': 'v0.99.0+recorded', 'install': 'v0.99.2+installed'}

            needed = {n:v for n,v in params.items() if n in dict(runtime.message_arguments(text))}
            result = runtime.render_message(text, needed, locale)
            for name, kind in runtime.message_arguments(text):
                if kind == 'simple':
                    assert str(params[name]) in result, (locale,key,count,name,'render lost parameter')
            assert set(re.findall(r'`([^`]+)`', result)) == expected_literals
            assert not (set(result) & CONTROLS)

            if plural:

            rendered += [f'## {key} | count={count} | CLDR={category}', '', result, '']
            render_count += 1
