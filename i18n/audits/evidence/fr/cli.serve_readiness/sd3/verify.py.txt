import json, re, hashlib, subprocess, unicodedata
from pathlib import Path
from local_operator.i18n import runtime as rt
from rich.cells import cell_len
root=Path.cwd()
base=Path('local_operator/i18n/catalogues')
ns='cli.serve_readiness'
def strict_pairs(pairs):
    out={}
    for k,v in pairs:
        assert k not in out, ('duplicate',k)
        out[k]=v
    return out
load=lambda p:json.loads(p.read_bytes(),object_pairs_hook=strict_pairs)
en=load(base/'en'/(ns+'.json'))
fr=load(base/'fr'/(ns+'.json'))
ctx=load(base/'en'/(ns+'.context.json'))
assert len(en)==len(fr)==len(ctx)==15
assert en.keys()==fr.keys()==ctx.keys()
def args(nodes):
    found=set()
    for n in nodes:
        if isinstance(n,rt._Arg):
            found.add((n.name,n.kind,n.skeleton))
            for _,body in n.parts: found.update(args(body))
    return found
def literals(nodes):
    txt=''.join(n.value for n in nodes if isinstance(n,rt._Text))
    return sorted(re.findall(r'`[^`]+`|</?[^>]+>',txt))
def contract(src,dst):
    assert args(src)==args(dst)
    assert literals(src)==literals(dst)
    sa={n.name:n for n in src if isinstance(n,rt._Arg) and n.parts}
    da={n.name:n for n in dst if isinstance(n,rt._Arg) and n.parts}
    assert sa.keys()==da.keys()
    for name,snode in sa.items():
        dnode=da[name]
        sb,db=dict(snode.parts),dict(dnode.parts)
        if snode.kind=='plural': assert set(db)=={'one','many','other'}
        else: assert sb.keys()==db.keys()
        for selector,body in db.items(): contract(sb.get(selector,sb['other']),body)

    has_plural=bool(rt.plural_selectors(fr[key]))
    plural_count+=int(has_plural)
    print('PASS',key,'args=',rt.message_arguments(fr[key]),'categories=',rt.plural_selectors(fr[key]))
    for count in ((0,1,2,1000000,1500000) if has_plural else (1,)):

assert plural_count==6

    print('RETAINED EXACT JSON LINE',k,hashlib.sha256(a[0]).hexdigest())
