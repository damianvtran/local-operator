"""Translator-only mechanical validation; not an independent audit or UI proof.
Run using the primary venv, PYTHONPATH pointing at the assigned worktree,
and an isolated HOME/config. Only the official checker's namespace/locale
enumeration is narrowed; catalogue, context, ICU and rendering code is real.
"""
from pathlib import Path
import hashlib
import json
import os
import unicodedata
from unittest.mock import patch
import local_operator
from local_operator.i18n import catalogues, messages, runtime, resolve
from rich.cells import cell_len
from rich.text import Text
from scripts.i18n import check

root = Path(__file__).resolve().parents[6]
ns = 'tui.provider_state'
assert Path(local_operator.__file__).resolve() == root / 'local_operator/__init__.py'
assert not any(k.startswith('CMUX_') for k in os.environ)
print('module:', local_operator.__file__)
print('HOME:', os.environ['HOME'])
print('config:', os.environ['LOCAL_OPERATOR_CONFIG_DIR'])
print('scope: en/fr tui.provider_state only; no eligibility patch or global checks')
base = root / 'local_operator/i18n/catalogues'
for locale, expected in [('en', '113c0264cd259c7a946e89cbd8a80577bfbdd7d8d2fadc9ac14a7e80da7f7912')]:
    assert hashlib.sha256((base / locale / f'{ns}.json').read_bytes()).hexdigest() == expected
assert hashlib.sha256((base / 'en' / f'{ns}.context.json').read_bytes()).hexdigest() == '67c6fda3895610bb57008f7fb97fce0f66f4b68cc9d23b70f87ab5d5abb7b7be'
real_namespaces = catalogues.namespaces
with patch.object(catalogues, 'namespaces', side_effect=lambda locale: (ns,) if locale in ('en', 'fr') else ()), patch.object(catalogues, 'locales', return_value=('en', 'fr')):
    problems = check.check_catalogues()
    print('official check.check_catalogues():', problems)
    assert not problems
    en = catalogues.load_catalogue('en', ns)
    fr = catalogues.load_catalogue('fr', ns)
    context = catalogues.context('en', ns)
    assert len(en) == len(fr) == len(context) == 5
    assert list(en) == list(fr) == list(context)
    assert resolve.eligible_locale('fr') == 'fr'
    for key, source in en.items():
        target = fr[key]
        assert runtime.message_arguments(source) == runtime.message_arguments(target) == ()
        assert runtime.plural_selectors(source) == runtime.plural_selectors(target) == ()
        assert unicodedata.normalize('NFC', target) == target
        assert not any(unicodedata.bidirectional(c) in ('LRE','RLE','LRO','RLO','PDF','LRI','RLI','FSI','PDI') for c in target)
        for locale, value in [('en', source), ('fr', target)]:
            assert runtime.render_message(value, {}, locale) == value
            result = messages.render_record(key, {}, locale=locale)
            assert result.text == value and result.source_locale == locale
        pseudo = '⟦' + target + ' ···⟧'
        assert runtime.message_arguments(pseudo) == ()
        assert runtime.render_message(pseudo, {}, 'fr') == pseudo
        print(key, '| en:', source, '| fr:', target, '| args=0 plural/select=0 NFC=yes | cells en/fr/pseudo:', cell_len(source), cell_len(target), cell_len(pseudo))
        # Context formula only: no invented maximum and no real pane assertion.
        for provider in ('openrouter', 'ollama'):
            budget = max(31 - cell_len(provider) - 4, 3)
            en_fit = Text(source)
            fr_fit = Text(target)
            en_fit.truncate(budget, overflow='ellipsis')
            fr_fit.truncate(budget, overflow='ellipsis')
            print('  diagnostic pane=31 synthetic-id=', provider, 'budget=', budget, 'en-fit=', en_fit.plain, 'fr-fit=', fr_fit.plain)
    # Negative control proves the scoped official checker actually reads fr.
    original_load = catalogues.load_catalogue
    def mutated_load(locale, namespace):
        values = original_load(locale, namespace)
        if locale == 'fr':
            values = dict(values)
            values[next(iter(values))] += ' {unexpected}'
        return values
    with patch.object(catalogues, 'load_catalogue', side_effect=mutated_load):
        errors = check.check_catalogues()
        assert any('arguments' in error and 'unexpected' in error for error in errors)
        print('negative control rejected added ICU argument:', errors)
print('RESULT: mechanical validation successful; status translated ONLY; independent A/B audit pending')
print('LIMITS: pure catalogue/ICU/render API and Rich width diagnostics; no settings/native/visual/network-isolation/release claim')
