# ------------------------------------------- 7. tokens, NFC, typography
TOKENS = {
    "tui.notifications.banner_digest_body": ["/notifications"],
    "tui.notifications.refused_argument": ["/notifications", "read"],
    "tui.notifications.pointer_unread": ["/unread"],

for key, tokens in TOKENS.items():
    for token in tokens:
        if token not in tg[key]:
            fail(f"{key}: literal token {token!r} missing from the target")
if " a " not in tg["tui.notifications.ask_answer_hint"]:
    fail("ask_answer_hint: key letter `a` missing")
if "f" not in tg["tui.notifications.empty_caught_up_body"]:
    fail("empty_caught_up_body: filter key `f` missing")
ok("literal tokens (/notifications, /unread, `/unread read`, read, y/n, a, f) present byte-exact")

BIDI = [0x202A, 0x202B, 0x202C, 0x202D, 0x202E, 0x2066, 0x2067, 0x2068, 0x2069, 0x200E, 0x200F, 0x061C]
for key, text in tg.items():
    if unicodedata.normalize("NFC", text) != text:
        fail(f"{key}: not NFC")
    bad = [hex(ord(c)) for c in text if ord(c) in bidi_set()] if False else [
        f"U+{ord(ch):04X}" for ch in text if ord(ch) in set(BIDI)
    ]
    if bad:
        fail(f"{key}: manual bidi controls {bad}")
    if "'" in text:
        fail(f"{key}: straight apostrophe U+0027 (fr guide requires ’)")
ok("NFC holds; no manual bidi controls; no straight apostrophes")

NARROW = {"\u202f": "U+202F", "\u00a0": "U+00A0", " ": "U+0020"}
for key, text in tg.items():
    for i, ch in enumerate(text):
        if ch in ":;!?":
            before = text[i - 1] if i else ""
            if before and before not in "\u202f\u00a0":
                fail(f"{key}: char {ch!r} at {i} preceded by {NARROW.get(before, repr(before))}, not U+202F")
    for i, ch in enumerate(text):
        if ch == "«":
            if text[i + 1] != "\u202f":
                fail(f"{key}: « not followed by U+202F ({NARROW.get(text[i+1], repr(text[i+1]))})")
        if ch == "»":
            if text[i - 1] != "\u202f":
                fail(f"{key}: » not preceded by U+202F ({NARROW.get(text[i-1], repr(text[i-1]))})")
ok("fr typography: U+202F before : ; ! ? and inside « … »")

for key, text in tg.items():
    if "  " in text:
        fail(f"{key}: double space")
    if text != text.strip():
        fail(f"{key}: leading/trailing whitespace")
ok("no double spaces, no stray whitespace")

# ------------------------------- 8. footer hint row cost, pinned function
render_src = (PIN / "local_operator/tui/notifications_render.py").read_text("utf-8")
tree = ast.parse(render_src)
fn = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == "hint_row")
ns: dict = {"cell_len": cell_len, "Sequence": list}
exec(compile(ast.Module(body=[fn], type_ignores=[]), "pinned:hint_row", "exec"), ns)
hint_row = ns["hint_row"]

KEYS = {
    "esc": "hint_back",
    "o": "hint_conversation",
    "a": "hint_answer",
    "e": "hint_dismiss",
    "E": "hint_dismiss_all",
    "u": "hint_unread",
    "u2": "hint_read",
    "space": "hint_select",
    "f": "hint_filter",
    "z": "hint_undo",
    "\u21b5": "hint_open",
}
# list-mode order from notifications_view._hint_labels (non-ask row, undo live)
ORDER = ["\u21b5", "e", "u2", "space", "o", "f", "z", "esc"]
for width in (80, 100, 120):
    row = {}
    for loc, blob in (("en", en), (LOCALE, tg)):
        labels = [(k, blob[f"{NS}.{KEYS[k]}"] if k != "u2" else blob[f"{NS}.hint_read"]) for k in ORDER]
        kept = hint_row(labels, width - 1)
        row[loc] = (len(kept), [k for k, _ in kept],
                    sum(cell_len(f"{k} {v}") for k, v in kept) + 2 * max(0, len(kept) - 1) + 1)
    print(f"HINT width={width}: en kept {row['en'][0]}/{len(ORDER)} cost {row['en'][2]} "
          f"{row['en'][1]} | fr kept {row[LOCALE][0]}/{len(ORDER)} cost {row[LOCALE][2]} "
          f"{row[LOCALE][1]}")

      f"'{core_slash.get('wire.slash.notifications_page') or page['wire.slash.notifications_page.help']}'")

print("-- negative controls (each MUST fail closed) --")


def control(name: str, fnc) -> None:
    try:
        caught, detail = fnc()
    except Exception as exc:  # noqa: BLE001
        caught, detail = True, f"raised {type(exc).__name__}: {exc}"
    print(("PASS" if caught else "FAIL") + f" negative control: {name} -> {detail}")
    if not caught:
        failures.append(f"negative control not caught: {name}")


def ctl_renamed_argument():
    bad = tg["tui.notifications.header_counts"].replace("{unread,", "{unreads,", 1).replace(
        " unread}}", " unreads}}", 1
    )
    problem = chk._arguments_problem(
        rt.message_arguments(en["tui.notifications.header_counts"]),
        rt.message_arguments(bad),
        translated_source=bad, locale=LOCALE, plural=plural,
    )
    return problem is not None, problem


def ctl_dropped_placeholder():
    src = tg["tui.notifications.updates_since"]
    bad = src.replace("{time}", "")
    assert "{time}" not in bad, "rig: the control did not actually drop every placeholder"
    leaked = "14:05" in rt.render_message(bad, {"count": 3, "time": "14:05"}, LOCALE)
    problem = chk._arguments_problem(
        rt.message_arguments(en["tui.notifications.updates_since"]),
        rt.message_arguments(bad),
        translated_source=bad, locale=LOCALE, plural=plural,
    )
    return (problem is not None) and not leaked, f"contract: {problem}; value leaked={leaked}"


def ctl_wrong_category():
    bad = tg["tui.notifications.dismissed"].replace("many {", "few {", 1)
    plist: list[str] = []
    chk._check_plural_categories(plist, LOCALE, NS, "dismissed", bad, plural, LOCALE)
    return bool(plist), plist


def ctl_missing_category():
    bad = tg["tui.notifications.dismissed"].replace(
        "many {# de mises à jour ignorées} ", "", 1
    )
    seen_ok = False
    try:
        rt.render_message(bad, {"count": 1000000}, LOCALE)
        seen_ok = True
    except rt.MessageFormatError as exc:
        return True, f"runtime refused the `many` render: {exc}"
    selectors = set(rt.plural_selectors(bad)[0])
    missing = sorted(fr_categories - selectors)
    return (not seen_ok) or bool(missing), f"selectors {sorted(selectors)}; missing {missing}"


def ctl_not_nfc():
    bad = "mise a\u0300 jour"
    return unicodedata.normalize("NFC", bad) != bad, "decomposed è detected"


def ctl_bidi():
    bad = "\u202e" + tg["tui.notifications.title"]
    return any(ord(c) in set(BIDI) for c in bad), "U+202E detected"


def ctl_literal_lost():
    bad = tg["tui.notifications.unread_mark_these"].replace("/unread read", "/unread")
    return "/unread read" not in bad, "typed command no longer byte-exact"


def ctl_overflow():
    bad = tg["tui.notifications.chip_high"] + " haute"
    return cell_len(bad) > BUDGETS["tui.notifications.chip_high"], f"{cell_len(bad)} > 6"


def ctl_typography():
    bad = tg["tui.notifications.marked_read"].replace("\u202f:", ":")
    return "\u202f:" not in bad and ":" in bad, "plain space before colon no longer matches the rule"


for name, fnc in (
    ("renamed argument {unread}->{unreads}", ctl_renamed_argument),
    ("dropped {time} placeholder", ctl_dropped_placeholder),
    ("wrong plural category `few`", ctl_wrong_category),
    ("missing `many` category", ctl_missing_category),
    ("non-NFC value", ctl_not_nfc),
    ("manual bidi control", ctl_bidi),
    ("typed command literal lost", ctl_literal_lost),
    ("overflow past maxCells", ctl_overflow),
    ("plain space before a colon", ctl_typography),
):
    control(name, fnc)
