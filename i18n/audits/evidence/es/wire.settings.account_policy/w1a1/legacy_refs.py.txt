#!/usr/bin/env python3
"""Does `legacy.help` name the policy labels verbatim, in en and in es?

The legacy help text refers to three policy labels by name ("Most headroom",
"Fair spread", "Reset soonest" in en). A translation that renames a label in
one key and not the other sends the reader to a control that does not exist,
so the reference must be byte-identical to the label value it points at.

Read-only.
"""
import json

BASE = "local_operator/i18n/catalogues"
NS = "wire.settings.account_policy"
en = json.load(open(f"{BASE}/en/{NS}.json", encoding="utf-8"))
es = json.load(open(f"{BASE}/es/{NS}.json", encoding="utf-8"))

pairs = [
    ("reset_soonest", "Reinicio más próximo"),
    ("headroom", "Más margen"),
    ("fair", "Reparto equitativo"),
]
print("== label references inside legacy.help ==")
for k, es_expected in pairs:
    lk = f"{NS}.{k}.label"
    in_en = en[lk] in en[f"{NS}.legacy.help"]
    in_es = es[lk] in es[f"{NS}.legacy.help"]
    print(f"{k}: en label {en[lk]!r} named in en legacy.help -> {in_en}")
    print(f"{k}: es label {es[lk]!r} named in es legacy.help -> {in_es}")
    print(f"{k}: es label matches the reading auditor's expectation {es_expected!r} -> {es[lk] == es_expected}")

print(f"label cell widths vs declared maxCells 29: "
      + ", ".join(f"{k}={len(es[f'{NS}.{k}.label'])}" for k, _ in pairs))
