import os, sys, json, hashlib, subprocess, runpy, re
from pathlib import Path
repo=Path.home()/'local-operator'
root=Path(__file__).resolve().parent
rev='9c979a03d551020ce6ff1b16c8066aa69ce7bb2a'
ns='wire.errors.session_host'
fixture=root/'fixture'
source_prefix='local_operator/i18n/catalogues/'
candidate=Path('/Users/damian/.local-operator/sessions/056852035b33/scratchpad/host-copy-es/wire.errors.session_host.json')
def blob(path): return subprocess.check_output(['git','-C',str(repo),'show',f'{rev}:{path}'])
def put(path, data):
    path.parent.mkdir(parents=True, exist_ok=True); path.write_bytes(data)
def sha(data): return hashlib.sha256(data).hexdigest()
files=['scripts/i18n/check.py','scripts/i18n/login.py']+[f'local_operator/i18n/{x}.py' for x in ['__init__','catalogues','runtime','format','messages','resolve']]
files += subprocess.check_output(['git','-C',str(repo),'ls-tree','-r','--name-only',rev,'local_operator/i18n/data']).decode().splitlines()
for p in files: put(fixture/p,blob(p))
enpath=source_prefix+'en/'+ns+'.json'; ctxpath=source_prefix+'en/'+ns+'.context.json'; espath=source_prefix+'es/'+ns+'.json'
enbytes=blob(enpath); ctxbytes=blob(ctxpath); targetbytes=candidate.read_bytes()
assert sha(enbytes)=='b7d6561be15577c118ec333586f7c809e65471c5dfd0dd839abee700b2ed1ba0'
assert sha(ctxbytes)=='d26acd5a6357a0c54bbd5ed409b4ad2c060ee52b9bd76670679397c4b464f724'
for p,b in [(enpath,enbytes),(ctxpath,ctxbytes),(espath,targetbytes)]: put(fixture/p,b)

check=runpy.run_path(str(fixture/'scripts/i18n/check.py'))
cat=check['catalogues']; runtime=check['msg_runtime']
assert Path(cat.__file__).is_relative_to(fixture)
assert Path(runtime.__file__).is_relative_to(fixture)

# Strict duplicate-key rejection supplements ordinary JSON parsing.
def pairs(items):
    obj={}
    for k,v in items:
        assert k not in obj, f'duplicate key: {k}'
        obj[k]=v
    return obj
en=json.loads(enbytes, object_pairs_hook=pairs); es=json.loads(targetbytes, object_pairs_hook=pairs); context=json.loads(ctxbytes, object_pairs_hook=pairs)
assert set(en)==set(es)==set(context) and len(en)==3
from rich.cells import cell_len
for k in en:
    assert re.findall(r'\{[^{}]*\}',en[k])==re.findall(r'\{[^{}]*\}',es[k])==[]
    assert re.findall(r'`[^`]*`',en[k])==re.findall(r'`[^`]*`',es[k])==[]
    assert not re.search(r'<[^>]+>', en[k]+es[k])
    assert en[k].count('logs/session-host.log')==es[k].count('logs/session-host.log')
    assert en[k].count('Local Operator')==es[k].count('Local Operator')
    print('PASS',k,'keys/placeholders/markup/literals; cells en=',cell_len(en[k]),'es=',cell_len(es[k]),'; no declared maxCells/maxChars=',not any(x in context[k] for x in ['maxCells','maxChars']))
print('PASS: all 3 keys; valid JSON; empty placeholder/ICU/markup sets; exact path/product literal parity')
print('LIMIT: static cell measurements only; no rendered fit proof; full CLI/full suite not run')
