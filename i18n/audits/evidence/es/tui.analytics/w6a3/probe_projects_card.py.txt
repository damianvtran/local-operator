import os, sys

cases={"fresh,2 live":dict(age=7200,stale=False,live=2),"stale,1 live":dict(age=86400*3,stale=True,live=1),"fresh,12 live":dict(age=7200,stale=False,live=12),"fresh 99d? 10d,1 live":dict(age=86400*10,stale=False,live=1)}
for name,c in cases.items():

    v["progress_stale"]=c["stale"]
    lines=render_project_board([v],now=NOW).text.plain.split("\n")
    print(f"{name:24}", repr(lines[-1]), '| facts:', repr(lines[-2]))
