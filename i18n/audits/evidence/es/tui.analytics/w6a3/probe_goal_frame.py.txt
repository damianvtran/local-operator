import asyncio, os, sys
sys.path.insert(0, os.getcwd())
from local_operator.i18n import text
text.set_active_locale("es")

GOAL="Migrar el servicio de pagos al nuevo proveedor y verificar cada cobro con el equipo de finanzas antes del cierre trimestral del año"
async def main():
    s=FakeSession(); s.arm_goal(GOAL); app=OperatorApp(lambda:_factory(s))
    async with app.run_test(size=(100,40)) as pilot:
        for _ in range(300):
            if app._session is not None: break
            await pilot.pause(); await asyncio.sleep(0.01)
        await pilot.pause(); app._cmd_goal("",lambda b,kind="info":None); await pilot.pause()

except BaseException as e: print("ERR",e)
