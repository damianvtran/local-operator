def complete(target,locale):
 problems=[]
 for k,v in target.items():
  want_select=[(n.name,set(dict(n.parts))) for n in nodes(rt.parse_message(en[k])) if n.kind=='select']
  got_select=[(n.name,set(dict(n.parts))) for n in nodes(rt.parse_message(v)) if n.kind=='select']
  if want_select!=got_select: problems.append(k+': select arm parity')
  for n in nodes(rt.parse_message(v)):
   if n.kind=='plural':
    if set(dict(n.parts)) != set(fmt.plural_categories(locale)): problems.append(k+': missing/extra CLDR categories')
    for arm,part in n.parts:
     if not any(isinstance(x,rt._Hash) or (isinstance(x,rt._Arg) and x.name==n.name) for x in part): problems.append(k+': missing count in '+arm)
 return problems
