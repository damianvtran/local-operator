"""Independent B audit scoped mechanical/render evidence, not native UI proof."""
import hashlib, io, json, pathlib, re, sys, unicodedata
from unittest.mock import patch
from rich.console import Console
from rich.text import Text
from rich.cells import cell_len
from local_operator.i18n import catalogues, runtime
from scripts.i18n import check
ROOT=pathlib.Path(sys.argv[1]).resolve()
NS='tui.model_persistence'
print('Interpreter:',sys.executable)
for module in (catalogues,runtime,check):
    print('Module:',module.__name__,module.__file__)
    assert pathlib.Path(module.__file__).resolve().is_relative_to(ROOT)
paths={'en':ROOT/f'local_operator/i18n/catalogues/en/{NS}.json','context':ROOT/f'local_operator/i18n/catalogues/en/{NS}.context.json','ru':ROOT/f'local_operator/i18n/catalogues/ru/{NS}.json'}
expected={'en':'4a8206d8eee6b97ed175d85941864ed9b3a2f9f8b6cf1889e091a9e2e2c62bff','context':'b1add11b7a553f3f17da185584b5936f388817c7827e07a84bccc2fa996331bc','ru':'9d08bddeee59461ae1a9f27de2839c39dc7b0cad17b6171e26fd7cde82946940'}
def hashes(): return {k:hashlib.sha256(p.read_bytes()).hexdigest() for k,p in paths.items()}
print('BEFORE hashes:',json.dumps(hashes(),sort_keys=True)); assert hashes()==expected
# Restrict enumeration only; leave official catalogue/context/ICU validation unchanged.
with patch.object(catalogues,'namespaces',return_value=(NS,)), patch.object(catalogues,'locales',return_value=('en','ru')):
    problems=check.check_catalogues()
    print('Official scoped check_catalogues en/ru:',problems); assert not problems
    original=catalogues.load_catalogue
    def missing(locale,namespace):
        data=original(locale,namespace)
        return {k:v for k,v in data.items() if not (locale=='ru' and k==NS+'.saved')}
    with patch.object(catalogues,'load_catalogue',side_effect=missing):
        negative=check.check_catalogues(); print('Missing-key negative:',negative); assert any('missing key' in p for p in negative)
    def wrongarg(locale,namespace):
        data=dict(original(locale,namespace))
        if locale=='ru': data[NS+'.with_model']=data[NS+'.with_model'].replace('{label}','{wrong_label}')
        return data
    with patch.object(catalogues,'load_catalogue',side_effect=wrongarg):
        negative=check.check_catalogues(); print('Wrong-arg negative:',negative); assert negative
    print('Restored positive:',check.check_catalogues()); assert not check.check_catalogues()
en=catalogues.load_catalogue('en',NS); ru=catalogues.load_catalogue('ru',NS)
assert list(en)==list(ru) and len(ru)==4
commands={NS+'.persist':'/model default',NS+'.settings':'/settings',NS+'.saved':'/model saved'}
for key in en:
    args_en=runtime.message_arguments(en[key]); args_ru=runtime.message_arguments(ru[key])
    assert args_en==args_ru
    assert not runtime.plural_selectors(ru[key])
    assert not re.search(r',\s*(plural|select|selectordinal)\s*,',en[key]+ru[key])
    assert unicodedata.normalize('NFC',ru[key])==ru[key]
    assert re.findall(r'`[^`]*`|</?\w+[^>]*>|\[[^\]]*\]',en[key])==re.findall(r'`[^`]*`|</?\w+[^>]*>|\[[^\]]*\]',ru[key])
    if key in commands: assert ru[key].count(commands[key])==en[key].count(commands[key])==1
    if key.endswith('with_model'):
        assert ru[key].count('{label}')==ru[key].count('{routes}')==1
    print('KEY PASS',key,'args',args_ru,'plural/select=0','cells',cell_len(ru[key]),'NFC/protected tokens PASS')
# Deliberately synthetic pseudo expansion: only human text expands; placeholders and commands protected.
def pseudo(s):
    pattern=r'(/model default|/model saved|/settings|\{[^}]+\})'
    return ''.join(part if i%2 else ''.join(c*2 if c.isalpha() else c for c in part) for i,part in enumerate(re.split(pattern,s)))
for key,value in en.items():
    expanded=pseudo(value); assert runtime.message_arguments(expanded)==runtime.message_arguments(value)
    if key in commands: assert expanded.count(commands[key])==1
    print('PSEUDO',key,expanded)
# Plain Rich snapshots only. No NoticeBlock stylesheet, transcript padding, terminal native certificate.
cases=0
for locale,data in [('en',en),('ru',ru),('pseudo',{k:pseudo(v) for k,v in en.items()})]:
    routes=' · '.join(data[NS+'.'+suffix] for suffix in ('persist','saved','settings'))
    for label in ('p/m','provider/very-long-opaque-model-identifier-2026-11-22-experimental-128k'):
        text=runtime.render_message(data[NS+'.with_model'],{'label':label,'routes':routes},'en' if locale=='pseudo' else locale)
        assert text.count(label)==1 and text.count(routes)==1
        for width in (80,100,120):
            c=Console(width=width,color_system=None,force_terminal=False)
            lines=Text(text).wrap(c,width)
            assert all(cell_len(line.plain)<=width for line in lines)
            folds=[cmd for cmd in commands.values() if not any(cmd in line.plain for line in lines)]
            print('WRAP',locale,'label='+('short' if label=='p/m' else 'long'),'width',width,'lines',len(lines),'command_folds',folds)
            for line in lines: print('|'+line.plain+'|')
            cases+=1
    # Sole usable route when hosting absent/unknown: no /model route introduced.
    sole=runtime.render_message(data[NS+'.settings'],{},'en' if locale=='pseudo' else locale)
    assert '/model' not in sole and sole.count('/settings')==1
    for width in (80,100,120):
        c=Console(width=width,color_system=None,force_terminal=False)
        lines=Text(sole).wrap(c,width)
        print('SOLE ROUTE',locale,width,[line.plain for line in lines])
        cases+=1
print('Render cases:',cases,'(27); plain Rich, not styled/native or transcript-family proof')
print('AFTER hashes:',json.dumps(hashes(),sort_keys=True)); assert hashes()==expected
print('PASS: all four keys independently checked; zero plural/select; no catalogue writes')
