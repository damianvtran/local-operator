"""Author-only scoped mechanical checks; not an independent language audit.
Run with primary checkout .venv python, PYTHONPATH=this worktree, isolated HOME/config.
Official checker discovery is restricted in-process to en/ru + this namespace;
its catalogue/context/ICU/plural algorithms are not replaced. Negatives mutate
only the returned in-memory ru catalogue, never source or target files.
"""
import importlib.util
import json
import os
from pathlib import Path
import re
import sys
import unicodedata
from unittest.mock import patch
from io import StringIO
from rich.console import Console
from rich.cells import cell_len
from rich.text import Text
import local_operator
from local_operator.i18n import catalogues, runtime

ROOT = Path('/Users/damian/local-operator/.worktrees/i18n-provider-state-labels')
NS = 'tui.provider_state'
assert Path(local_operator.__file__).resolve() == ROOT / 'local_operator/__init__.py'
print('module proof:', local_operator.__file__)
assert Path.home() != Path('/Users/damian')
assert not any(k.startswith('CMUX_') for k in os.environ)
assert os.environ['LOCAL_OPERATOR_CONFIG_DIR'].startswith(str(Path.home()))
print('isolation: HOME/config redirected; CMUX_* absent; no app startup')
spec = importlib.util.spec_from_file_location('scoped_i18n_check', ROOT / 'scripts/i18n/check.py')
checker = importlib.util.module_from_spec(spec)
# Dataclasses resolve postponed annotations through sys.modules on import.
sys.modules[spec.name] = checker
spec.loader.exec_module(checker)
load = catalogues.load_catalogue
source = load('en', NS)
target = load('ru', NS)
context = catalogues.context('en', NS)
assert len(source) == len(target) == 5
assert list(source) == list(target) == list(context)
print('ordered five-key/context parity: PASS')

def check(candidate=None):
    def scoped_load(locale, namespace):
        return candidate if locale == 'ru' and candidate is not None else load(locale, namespace)
    with patch.object(catalogues, 'locales', return_value=('en', 'ru')), patch.object(catalogues, 'namespaces', return_value=(NS,)), patch.object(catalogues, 'load_catalogue', side_effect=scoped_load):
        return checker.check_catalogues()

problems = check()
print('OFFICIAL check_catalogues scoped en/ru tui.provider_state:', json.dumps(problems))
assert not problems
missing = dict(target)
missing.pop(next(iter(missing)))
problems = check(missing)
print('missing-key negative:', json.dumps(problems))
assert len(problems) == 1 and 'missing key' in problems[0]
extra_arg = dict(target)
extra_arg[next(iter(extra_arg))] += ' {injected}'
problems = check(extra_arg)
print('added-argument negative:', json.dumps(problems))
assert any('arguments' in p and 'injected' in p for p in problems)
assert not check()
print('restored official scoped control: PASS')

# Protected tokens are absent in these five English labels; do not invent any.
protected = re.compile(r'`[^`]*`|https?://\S+|%[sd]|</?\w+>|--[\w-]+')
for key in source:
    assert runtime.message_arguments(source[key]) == runtime.message_arguments(target[key]) == ()
    assert unicodedata.normalize('NFC', target[key]) == target[key]
    assert not any(unicodedata.category(c) == 'Cf' for c in target[key])
    assert protected.findall(source[key]) == protected.findall(target[key]) == []
    for locale, value in [('en', source[key]), ('ru', target[key])]:
        rendered = runtime.render_message(value, {}, locale)
        assert rendered == value
        rich = Text.from_markup(rendered)
        assert rich.plain == rendered
        assert rich.cell_len == cell_len(rendered)
        print('render', locale, key, repr(rendered), 'cells=', cell_len(rendered), 'args=0')
    # Artificial expansion is a diagnostic only: not a translated catalogue.
    pseudo = '⟦' + source[key] + ' · ' + source[key] + '⟧'
    print('pseudo diagnostic', key, repr(pseudo), 'cells=', cell_len(pseudo))
    for name, value in [('en', source[key]), ('ru', target[key]), ('pseudo', pseudo)]:
        stream = StringIO()
        console = Console(file=stream, width=31, force_terminal=False, color_system=None)
        console.print(Text('openrouter  ' + value))
        lines = stream.getvalue().splitlines()
        assert all(cell_len(line) <= 31 for line in lines)
        print('plain Rich illustrative 31-column row', name, 'line_cells=', [cell_len(line) for line in lines], 'lines=', repr(lines))
print('NFC/protected-token/zero-argument/ICU/render checks: PASS')
print('WIDTH LIMITATION: illustrative plain Rich only; no universal maxCells, no real settings/native geometry, font, no-wrap, integration, eligibility or language audit certification.')
print('AUTHOR STATUS: translated; independent A/B audits remain required.')
