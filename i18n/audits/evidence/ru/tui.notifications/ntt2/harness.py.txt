# Expected grammatical form per (key, plural argument, count). Probes the COUNTED
# WORD, not the numeral, so ru's NBSP thousands grouping cannot mask a case error.
# Category facts used: 1234 % 10 == 4 and 1234 % 100 == 34 -> `few` (not `many`).
def ru_category(n: int) -> str:
    """CLDR ru cardinal category for an integer count."""
    if n % 10 == 1 and n % 100 != 11:
        return "one"
    if n % 10 in (2, 3, 4) and n % 100 not in (12, 13, 14):
        return "few"
    return "many"

# ------------------------------------------- 6. typography and text controls
say("")
say("== typography, NFC and direction controls")
CTRLS = ["\u202a", "\u202b", "\u202c", "\u202d", "\u202e", "\u2066", "\u2067", "\u2068", "\u2069",
         "\u200e", "\u200f", "\u061c"]
bad_nfc, bad_ctrl, bad_punct, bad_space, trailing = [], [], [], [], []
for key, value in ru.items():
    if unicodedata.normalize("NFC", value) != value:
        bad_nfc.append(key)
    if any(c in value for c in CTRLS):
        bad_ctrl.append(key)
    if re.search(r"\s+[!:;?]", value) or "!!" in value or value.rstrip().endswith("!"):
        bad_punct.append((key, value))
    if "  " in value:
        bad_space.append((key, value))
    if value != value.strip():
        trailing.append(key)
    if "\u00ab" in value or "\u00bb" in value:
        if value.count("\u00ab") != value.count("\u00bb"):
            bad_punct.append((key, "unbalanced guillemets: " + value))
    if re.search(r"[\"“”]", value):
        bad_punct.append((key, "straight/curly quote in ru prose: " + value))

say("   NFC violations: %d %s" % (len(bad_nfc), bad_nfc))
if bad_nfc:
    fail("NFC violations: %s" % bad_nfc)
say("   bidi/format controls: %d %s" % (len(bad_ctrl), bad_ctrl))
if bad_ctrl:
    fail("manual bidi/format controls: %s" % bad_ctrl)
say("   punctuation problems (space before !?;:, !, mixed quotes): %d" % len(bad_punct))
for k, v in bad_punct:
    say("      %-22s %r" % (k.split(".", 2)[-1], v))
if bad_punct:
    fail("punctuation problems: %s" % [k for k, _ in bad_punct])
say("   double spaces: %d %s" % (len(bad_space), [k for k, _ in bad_space]))
if bad_space:
    fail("double spaces: %s" % [k for k, _ in bad_space])
say("   leading/trailing whitespace: %d %s" % (len(trailing), trailing))
if trailing:
    fail("stray padding spaces: %s" % trailing)

    worst, worst_params = max(measured, key=lambda pair: pair[0])
    en_w = cell_len(msg_runtime.render_message(en[key], worst_params, "en"))
    verdict = "OVER" if worst > budget else "ok"
    say("   %-22s budget=%-3s en=%-3d ru=%-3d %-4s ratio=%.2f worst=%r" % (
        key.split(".", 2)[-1], budget, en_w, worst, verdict,
        (worst / en_w) if en_w else 0.0,
        msg_runtime.render_message(ru[key], worst_params, LOCALE)))
    if worst > budget:
        over.append((key, budget, worst))
for key, budget, worst in over:
    fail("width: %s renders %d cells against maxCells %d" % (key, worst, budget))

ncfile = tmp / "catalogues" / LOCALE / (NS + ".json")
pristine = ncfile.read_text(encoding="utf-8")
orig_root = catalogues.catalogue_root()


def mutate(short: str, value: str) -> None:
    data = json.loads(pristine)
    data[NS + "." + short] = value
    ncfile.write_text(json.dumps(data, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def ru_problems() -> list[str]:
    catalogues.catalogue_root = lambda: tmp / "catalogues"  # type: ignore[assignment]
    try:
        return [p for p in check_catalogues() if p.startswith(LOCALE + "/" + NS + ":")]
    finally:
        catalogues.catalogue_root = lambda: orig_root  # type: ignore[assignment]


# NC1: renamed placeholder
mutate("confirm_dismiss_session", "Скрыть {name} из «{source}»? y/n")
nc = ru_problems()
say("   NC1 renamed {count} -> %d problem(s); first=%r" % (len(nc), nc[:1]))
if not any("confirm_dismiss_session" in p and "argument" in p for p in nc):
    fail("NC1 did not fail closed: a renamed placeholder passed")

# NC2: missing ru category (completeness rule)
mutate("asks_stay", "{count, plural, one {# запрос останется до ответа.} "
                    "many {# запросов останутся до ответа.} other {# запроса останутся до ответа.}}")
nc2 = _number_upgrade_problem(json.loads(ncfile.read_text(encoding="utf-8"))[NS + ".asks_stay"],
                              "count", LOCALE, plural_table)
say("   NC2 dropped `few` -> completeness rule says: %r" % (nc2,))
if not (nc2 and "few" in nc2):
    fail("NC2 did not fail closed: a missing `few` branch passed")

# NC3: empty branch
mutate("dismissed", "Скрыто {count, plural, one {} few {# обновления} many {# обновлений} "
                    "other {# обновления}}.")
nc3 = _number_upgrade_problem(json.loads(ncfile.read_text(encoding="utf-8"))[NS + ".dismissed"],
                              "count", LOCALE, plural_table)
say("   NC3 empty branch -> completeness rule says: %r" % (nc3,))
if not nc3:
    fail("NC3 did not fail closed: an empty plural branch passed")

# NC4: extra key (parity)
ncfile.write_text(pristine, encoding="utf-8")
data = json.loads(pristine)
data[NS + ".not_a_key"] = "лишнее"
ncfile.write_text(json.dumps(data, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
nc = ru_problems()
say("   NC4 extra key -> %d problem(s); first=%r" % (len(nc), nc[:1]))
if not any("unknown key" in p for p in nc):
    fail("NC4 did not fail closed: an extra key passed")

# NC5: drifted command literal (this script's token check)
ncfile.write_text(pristine, encoding="utf-8")
dropped = "Отчёты о завершении сессий перенесены в /уведомления."
say("   NC5 drifted /unread -> token check sees the literal missing: %s"
    % ("/unread" not in dropped))
if "/unread" in dropped:
    fail("NC5 control text unexpectedly still holds the literal")

# NC6: over-budget value (this script's width check)
wide = "видели в чате, когда вы были здесь"
say("   NC6 over-budget value -> cell_len=%d vs maxCells 12 -> over=%s"
    % (cell_len(wide), cell_len(wide) > 12))
if cell_len(wide) <= 12:
    fail("NC6 control value is not over budget")

# NC7: space before punctuation (this script's typography check)
probe = "Готово !"
say("   NC7 space-before-punctuation detected: %s"
    % bool(re.search(r"\s+[!:;?]", probe)))
if not re.search(r"\s+[!:;?]", probe):
    fail("NC7 punctuation detector did not fire")

# NC8: manual bidi control (this script's control scan)
probe = "Скрыто \u202eобновление\u202c."
say("   NC8 manual bidi control detected: %s" % any(c in probe for c in CTRLS))
if not any(c in probe for c in CTRLS):
    fail("NC8 bidi detector did not fire")

ncfile.write_text(pristine, encoding="utf-8")
shutil.rmtree(tmp, ignore_errors=True)

say("")
say("FAILURES=%r" % (failures,))
say("RESULT: %s (%d failure(s))" % ("PASS" if not failures else "FAIL", len(failures)))
sys.exit(1 if failures else 0)
