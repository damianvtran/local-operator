"""Setup preflight only: existing hermetic fixture, fail-closed networking, no frames."""
import os, sys, json, hashlib, inspect, asyncio, socket
from pathlib import Path
OUT=Path(os.environ['PREFLIGHT_OUT'])
assert Path.home()==OUT/'iso'
assert 'NO_COLOR' not in os.environ
assert not any(k.startswith('CMUX_') for k in os.environ)
assert os.environ['TERM']=='xterm-256color'
attempts=[]
class NetworkForbidden(RuntimeError): pass
def audit(event,args):
    if event in ('socket.connect','socket.getaddrinfo','socket.gethostbyname','socket.gethostbyaddr','socket.sendto'):
        attempts.append(event)
        raise NetworkForbidden(event)
sys.addaudithook(audit)
# Negative control reaches the lowest connection seam, not an actual network.
with socket.socket() as probe:
    try: probe.connect(('192.0.2.1',443))
    except NetworkForbidden: pass
    else: raise AssertionError('socket guard failed')
assert attempts==['socket.connect']
print('SOCKET_NEGATIVE_CONTROL blocked socket.connect before outbound',flush=True)
attempts.clear()
import pytest, httpx
mp=pytest.MonkeyPatch()
def deny_http(*args,**kwargs):
    attempts.append('httpx.send')
    raise NetworkForbidden('httpx.send')
async def deny_async_http(*args,**kwargs): return deny_http(*args,**kwargs)
mp.setattr(httpx.Client,'send',deny_http)
mp.setattr(httpx.AsyncClient,'send',deny_async_http)
try: httpx.get('https://example.invalid/')
except NetworkForbidden: pass
else: raise AssertionError('HTTP guard failed')
assert attempts==['httpx.send']
print('HTTP_NEGATIVE_CONTROL blocked httpx.send before transport',flush=True)
attempts.clear()
import local_operator
from tests.unit.tui.conftest import hermetic_tui_env
from tests.unit.tui.test_app_pilot import _SwitchableSession,_AccessController,_factory,_await_session
from local_operator.tui.app import OperatorApp
from local_operator.i18n.text import set_active_locale,active_locale
assert str(local_operator.__file__).startswith(os.environ['PYTHONPATH']+'/')
fixture=hermetic_tui_env.__wrapped__
source=inspect.getsource(fixture)
(OUT/'source-fixture.txt').write_text(source)
fixture(mp)
# Reuse the fixture-installed update no-op; wrap only to prove mount called it.
fixture_stub=OperatorApp._check_for_update
calls=[]
def counted_update(self):
    calls.append('update-worker')
    return fixture_stub(self)
mp.setattr(OperatorApp,'_check_for_update',counted_update)
config=OUT/'iso'/'config'
config.mkdir(parents=True,exist_ok=True)
mp.setenv('LOCAL_OPERATOR_CONFIG_DIR',str(config))
(config/'config.yml').write_text('version: 0.0.0\nvalues:\n  language: fr\n  hosting: openrouter\n')
async def main():
    session=_SwitchableSession()
    app=OperatorApp(lambda:_factory(session),provider_controller=_AccessController(stored=('openrouter','anthropic')))
    async with app.run_test(size=(100,30)) as pilot:
        await _await_session(app,pilot)
        set_active_locale('fr')
        for _ in range(4): await pilot.pause()
        assert active_locale()=='fr'
        assert calls==['update-worker'],calls
        assert attempts==[],attempts
        app._run_slash_command('/model')
        from scripts.visual_capture import save_capture
        save_capture(app, OUT/'first.svg')
        for _ in range(4): await pilot.pause()
        save_capture(app, OUT/'settled.svg')
        from local_operator.tui.app import NoticeBlock
        print('NOTICE_GEOMETRY '+json.dumps([{'text':b.text(),'size':str(b.size),'region':str(b.region),'virtual_size':str(b.virtual_size),'scrollbar':getattr(b,'show_vertical_scrollbar',None)} for b in app._transcript_view().blocks() if isinstance(b,NoticeBlock)]),flush=True)
        print('STARTUP_SETUP_PASS '+json.dumps({'module':local_operator.__file__,'fixture':inspect.getsourcefile(fixture)+':369 hermetic_tui_env','fixture_sha256':hashlib.sha256(source.encode()).hexdigest(),'update_calls':calls,'unexpected_network_attempts':attempts,'locale':active_locale(),'home':str(Path.home()),'config':str(config),'source_fixture_captured':True,'screenshots':0}),flush=True)
    assert not attempts,attempts
try: asyncio.run(main())
finally: mp.undo()
