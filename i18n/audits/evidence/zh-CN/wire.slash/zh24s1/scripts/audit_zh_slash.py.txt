def sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def sha256_file(path: Path) -> str:
    return sha256_bytes(path.read_bytes())


def cells(text: str) -> int:
    total = 0
    for ch in text:
        if unicodedata.combining(ch):
            continue
        total += 2 if unicodedata.east_asian_width(ch) in "WF" else 1
    return total


def write_json(name: str, payload: object) -> None:
    (B / name).write_text(
        json.dumps(payload, ensure_ascii=False, indent=2, sort_keys=False) + "\n",
        encoding="utf-8",
    )


def hard_tokens(text: str) -> dict[str, list[str]]:
    import re

    patterns = {
        "backtick_span": re.compile(r"`[^`\n]+`"),
        "long_flag": re.compile(r"(?<![\w/])--[a-zA-Z0-9][a-zA-Z0-9_-]*"),
        "angle_token": re.compile(r"<[a-zA-Z][^>\n]*>"),
        "url": re.compile(r"(?:https?|ssh)://[^\s，。；）)]+"),
        "path": re.compile(r"(?:~/[\w./-]+|/(?:bin|Users|Volumes|tmp|var|etc)/[\w./-]+)"),
        "percent_token": re.compile(r"%[sdif]"),
    }
    return {name: pattern.findall(text) for name, pattern in patterns.items()}


def tokens_equal(en: dict[str, list[str]], zh: dict[str, list[str]]) -> bool:
    return all(en[name] == zh[name] for name in en)

for key in EN:
    en_text = EN[key]
    zh_text = ZH[key]
    en_args = runtime.message_arguments(en_text)
    zh_args = runtime.message_arguments(zh_text)
    if en_args != zh_args:
        arg_failures.append({"key": key, "en": en_args, "zh": zh_args})
    selectors_en = runtime.plural_selectors(en_text)
    selectors_zh = runtime.plural_selectors(zh_text)
    if selectors_en or selectors_zh:
        plural_rows.append(
            {"key": key, "en": selectors_en, "zh": selectors_zh}
        )
        if selectors_en != selectors_zh:
            plural_failures.append(
                {"key": key, "en": selectors_en, "zh": selectors_zh}
            )
    try:
        en_rendered = runtime.render_message(en_text, {}, "en")
        zh_rendered = runtime.render_message(zh_text, {}, "zh-CN")
        if en_rendered != en_text or zh_rendered != zh_text:
            render_failures.append({"key": key, "en": en_rendered, "zh": zh_rendered})
    except Exception as exc:  # noqa: BLE001 - evidence records the failure
        render_failures.append({"key": key, "error": f"{type(exc).__name__}: {exc}"})
    en_tokens = hard_tokens(en_text)
    zh_tokens = hard_tokens(zh_text)
    if not tokens_equal(en_tokens, zh_tokens):
        token_failures.append({"key": key, "en": en_tokens, "zh": zh_tokens})
    en_cells = cells(en_text)
    zh_cells = cells(zh_text)
    width_rows.append(
        {
            "key": key,
            "en_cells": en_cells,
            "zh_cells": zh_cells,
            "delta_cells": zh_cells - en_cells,
            "ratio": round(zh_cells / en_cells, 3),
            "maxCells": None,
            "approx_help_budget_cells": 55,
            "fits_approx_help": zh_cells <= 55,
            "zh_shorter_than_en": zh_cells <= en_cells,
        }
    )
    import re

    for match in re.finditer(r"[,;:!?()\[\]\"]", zh_text):
        punct_rows.append(
            {"key": key, "char": match.group(), "offset": match.start()}
        )
    if re.search(r"[\u4e00-\u9fff] [A-Za-z0-9]|[A-Za-z0-9] [\u4e00-\u9fff]", zh_text):
        spacing_failures.append({"key": key, "text": zh_text})
    if unicodedata.normalize("NFC", zh_text) != zh_text:
        non_nfc.append(key)
    if re.search(r"[\u202a-\u202e\u2066-\u2069\u200e\u200f\u061c]", zh_text):
        bidi.append(key)
    if re.search(r"\d", zh_text):
        digit_rows.append({"key": key, "text": zh_text})

over_budget = [row for row in width_rows if row["zh_cells"] > 55]
materially_longer = [row for row in width_rows if row["zh_cells"] > row["en_cells"] + 4]
