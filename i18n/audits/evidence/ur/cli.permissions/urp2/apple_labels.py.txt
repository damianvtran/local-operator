import plistlib,glob,os,sys
targets=['Screen & System Audio Recording','Privacy & Security','System Settings','Files & Folders','Accessibility','Keychain Access','Access Control','Always Allow','Allow','Screen Recording']
def load(p):
    try: return plistlib.load(open(p,'rb'))
    except Exception as e: return None
files=glob.glob('/System/Library/ExtensionKit/Extensions/SecurityPrivacyExtension.appex/Contents/Resources/*.loctable')
files+=glob.glob('/System/Applications/System Settings.app/Contents/Resources/*.loctable')
files+=glob.glob('/System/Library/Frameworks/Security.framework/Versions/A/MachServices/SecurityAgent.bundle/Contents/Resources/*.loctable')
files+=glob.glob('/System/Library/Frameworks/Security.framework/Versions/A/MachServices/SecurityAgent.bundle/Contents/Resources/en.lproj/*.strings')
files+=glob.glob('/System/Library/CoreServices/Applications/Keychain Access.app/Contents/Resources/*.loctable')
files+=glob.glob('/System/Library/CoreServices/Applications/Keychain Access.app/Contents/Resources/en.lproj/*')
found={t:[] for t in targets}
langs=set()
for f in files:
    d=load(f)
    if not isinstance(d,dict): continue
    tables=d if f.endswith('.loctable') else {'en':d}
    if f.endswith('.loctable'): langs|=set(tables)
    en=tables.get('en') or {}
    if not isinstance(en,dict): continue
    for k,v in en.items():
        if isinstance(v,str) and v in targets: found[v].append(f"{os.path.basename(os.path.dirname(os.path.dirname(os.path.dirname(f))))}/{os.path.basename(f)}[en] key={k!r}")
for t,v in found.items(): print(t,'->',v[:3] if v else 'NOT FOUND')
print('ur in loctable languages:', 'ur' in langs, '| hi:', 'hi' in langs, '| n langs', len(langs))

def nibstr(p,needles):
    fs=[p] if os.path.isfile(p) else glob.glob(p+'/*')
    out=set()
    for f in fs:
        s=subprocess.run(['strings','-a',f],capture_output=True,text=True).stdout
        for n in needles:
            if n in s.splitlines(): out.add((os.path.relpath(f,'/System'),n))
    return out
print(nibstr(SA+'/Base.lproj/ConfirmAccess.nib',['Always Allow','Allow','Deny']))
