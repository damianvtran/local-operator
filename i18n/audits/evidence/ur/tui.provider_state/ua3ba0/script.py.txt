with patch.object(c,'locales',return_value=('en','ur')),patch.object(c,'namespaces',return_value=(NS,)):
    problems=check.check_catalogues(); print('OFFICIAL scoped catalogue/context/ICU',problems); assert not problems
    original=c.load_catalogue
    def missing(locale,namespace):
        value=dict(original(locale,namespace))
        if locale=='ur': value.pop(next(iter(value)))
        return value
    with patch.object(c,'load_catalogue',side_effect=missing):
        failures=check.check_catalogues(); print('NEGATIVE missing-key',failures); assert any('missing key' in x for x in failures)
    def extra(locale,namespace):
        value=dict(original(locale,namespace))
        if locale=='ur': value[next(iter(value))]+=' {unexpected}'
        return value
    with patch.object(c,'load_catalogue',side_effect=extra):
        failures=check.check_catalogues(); print('NEGATIVE added-argument',failures); assert any('arguments' in x for x in failures)
for key,text in ur.items():
    assert r.message_arguments(text)==r.message_arguments(en[key])==()
    assert r.plural_selectors(text)==() and r.plural_arguments(text)==()
    assert all(type(node).__name__=='_Text' for node in r.parse_message(text))
    assert unicodedata.normalize('NFC',text)==text
    assert not any(unicodedata.category(ch) in ('Cf','Cc') for ch in text)
    protected=r'`[^`]*`|<[^>]+>|\[[^\]]+\]|--[A-Za-z][\w-]*|[A-Za-z_][A-Za-z_0-9]*'
    assert re.findall(protected,text)==[] # No IDs/commands/tags in these human labels.
    raw=r.render_message(text,{},'ur'); assert raw==text
    assert Text(raw).plain==raw and Text.from_markup(raw).plain==raw
    projected=isolated_status_text(RenderedMessage(raw,'ur')); assert projected=='\u2067'+raw+'\u2069'
    assert Text(projected).plain==projected
    english=r.render_message(en[key],{},'en'); assert english==en[key]
    assert isolated_status_text(RenderedMessage(english,'en'))==english

    isolated=isolated_status_text(RenderedMessage(mixed,'ur'),[TechnicalSpan(len(raw)+1,len(mixed),'/provider')])
    assert '\u2066/provider\u2069' in isolated and isolated.startswith('\u2067') and isolated.endswith('\u2069')
    print('PASS',key,json.dumps(text,ensure_ascii=False),'zeroargs/no plural-select NFC/controlfree protected-empty plain/Rich English/pseudo RTL', 'cells',cell_len(raw),'codepoints',len(raw))
