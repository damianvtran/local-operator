"""Scoped author validation; not an independent linguistic or native RTL audit.
Run with the primary checkout interpreter, explicit worktree PYTHONPATH and
isolated HOME/config. Widths are diagnostic only, not shaping/layout proof.
"""
from pathlib import Path
import hashlib
import json
import re
import unicodedata
import importlib.util
from collections import Counter
from rich.cells import cell_len
from rich.console import Console
from rich.text import Text
from local_operator.i18n import catalogues, runtime

ROOT = Path('/Users/damian/local-operator/.worktrees/i18n-model-persist-notice')
NS = 'tui.model_persistence'
expected = {'en': '4a8206d8eee6b97ed175d85941864ed9b3a2f9f8b6cf1889e091a9e2e2c62bff', 'context': 'b1add11b7a553f3f17da185584b5936f388817c7827e07a84bccc2fa996331bc'}
for module in (catalogues, runtime):
    assert Path(module.__file__).resolve().is_relative_to(ROOT)
    print('MODULE', module.__name__, module.__file__)
spec = importlib.util.spec_from_file_location('scoped_check', ROOT / 'scripts/i18n/check.py')
check = importlib.util.module_from_spec(spec)
import sys
sys.modules[spec.name] = check
spec.loader.exec_module(check)
paths = {locale: ROOT / f'local_operator/i18n/catalogues/{locale}/{NS}.json' for locale in ('en', 'ur')}
paths['context'] = ROOT / f'local_operator/i18n/catalogues/en/{NS}.context.json'
hashes = {name: hashlib.sha256(path.read_bytes()).hexdigest() for name, path in paths.items()}
assert all(hashes[name] == value for name, value in expected.items())
en = catalogues.load_catalogue('en', NS)
ur = catalogues.load_catalogue('ur', NS)
context = json.loads(paths['context'].read_text())
assert len(ur) == 4 and list(en) == list(ur) == list(context)
controls = set(range(0x202A, 0x202F)) | set(range(0x2066, 0x206A)) | {0x200E, 0x200F, 0x061C}
for key, source in en.items():
    target = ur[key]
    assert catalogues.context_entry_problem(context[key]) is None
    src_args, dst_args = runtime.message_arguments(source), runtime.message_arguments(target)
    assert check.arguments_match(src_args, dst_args)
    assert all(kind == 'simple' for _, kind in dst_args)
    assert Counter(re.findall(r'\{(\w+)\}', source)) == Counter(re.findall(r'\{(\w+)\}', target))
    ast = runtime.parse_message(target)
    assert not any(getattr(node, 'kind', None) in ('plural', 'select') for node in ast)
    assert runtime.plural_selectors(target) == ()
    assert unicodedata.normalize('NFC', target) == target
    assert not any(ord(char) in controls for char in target)
    for token in ('/model default', '/model saved', '/settings'):
        assert source.count(token) == target.count(token)
    human = re.sub(r'/model default|/model saved|/settings|\{label\}|\{routes\}', '', target)
    assert not re.search(r'[A-Za-z]', human), 'Unexpected human English'
    print('KEY PASS', key, 'arguments=', dst_args, 'plural/select=0', 'cells=', cell_len(target))

routes = ' · '.join(runtime.render_message(ur[f'{NS}.{key}'], {}, 'ur') for key in ('persist', 'saved', 'settings'))
for label in ('openai/gpt-6', 'anthropic/claude-opus-5-5-20261122-opaque-provider-model-identity-long-0123456789'):
    result = runtime.render_message(ur[f'{NS}.with_model'], {'label': label, 'routes': routes}, 'ur')
    assert result == 'ماڈل: ' + label + ' — ' + routes
    assert result.count(label) == result.count(routes) == 1
    print('RENDER', repr(result))
    for width in (80, 100, 120):
        console = Console(width=width, force_terminal=False, color_system=None)
        lines = Text(result).wrap(console, width)
        print('WIDTH_DIAGNOSTIC', 'label_chars=', len(label), 'width=', width, 'line_cells=', [line.cell_len for line in lines])
for name, path in paths.items():
    assert hashlib.sha256(path.read_bytes()).hexdigest() == hashes[name]
print('HASHES', json.dumps(hashes, sort_keys=True))
print('PASS: four ordered keys; official catalogue/context/ICU/argument APIs; exact commands; opaque arguments once; zero plural/select arms; NFC; no bidi controls; short/long labels; frozen inputs unchanged')
print('NOT PROVEN: native shaping/fonts, styled NoticeBlock geometry, live integration, independent linguistic audit, eligibility or release clearance')
