#!/usr/bin/env python3
"""Read-only le28r2 verifier; run from the repository root with existing Python.

This verifies file integrity and history preservation, NOT linguistic correctness
or sanctioned read_public_copy access. No live config/session or browser is used.
"""
import hashlib
import json
import subprocess
from pathlib import Path

root = Path.cwd()
base = '9495f8a6c3bf2b5438ef5c33d7274cf519435dda'
old = json.loads(subprocess.check_output(['git','show',base+':i18n/ledger.json']))['entries']
new = json.loads((root/'i18n/ledger.json').read_text())['entries']
# Merge-stable, and append-honest: other PRs append their own ledger rows, and a
# text merge may place them BEFORE rows that existed at the audited base (observed:
# main's relay.asks rows landed ahead of the base's last five ur rows), so neither
# total length, row index nor "the last 24" is a stable handle.
#
# This PR also does NOT append a ledger row. i18n/ledger.json admits ONE row per
# (locale, namespace) at a given English source hash - tests/unit/i18n/test_hold_split.py
# enforces it ("a second row at the same hash is a duplicate, not history") - and the
# lost-evidence recovery re-judged the SAME English bytes, so the only legal shape is
# to keep the existing row and refresh it: the audit note is extended and the auditor
# field is replaced by the live recovery auditor. The prior row verbatim is preserved
# in each bundle's prior-ledger.json. So: (1) every base row must reappear, in order,
# as a subsequence of the current ledger with every field except note/auditor identical,
# its note only ever extended, and its auditor either untouched or replaced (counted);
# (2) exactly 24 rows are refreshed; (3) no row is appended by this PR; (4) each
# refreshed row is the pair's LAST row, at the current English hash, and no earlier row
# for that pair shares that hash.
nf=lambda e:{k:v for k,v in e.items() if k not in ('note','auditor')}
j=0;matched=set();changed=0;refreshed=[]
for a in old:
    while j<len(new) and not (nf(new[j])==nf(a) and new[j]['note'].startswith(a['note']) and j not in matched):
        j+=1
    assert j<len(new),('base row lost or altered',a['locale'],a['namespace'])
    matched.add(j); changed += new[j]['note']!=a['note']
    if new[j]['auditor']!=a['auditor']: refreshed.append(new[j])
    j+=1
assert len(matched)==len(old),(len(matched),len(old))
assert len(refreshed)==24,('rows refreshed',len(refreshed))
rows={(r['locale'],r['namespace']):r for r in refreshed}
assert len(rows)==24
sentence='lost-evidence re-audit supersedes'
for (locale,namespace),r in rows.items():
    # the supersession marker sits on the row whose evidence was lost, which is the
    # refreshed row for 15 pairs and an earlier row for the 9 whose later row was
    # audited at these bytes already; either way that pair has exactly one marker.
    assert sum(sentence in e['note'] for e in new if (e['locale'],e['namespace'])==(locale,namespace))==1,(locale,namespace)
    assert 'audits/evidence/%s/%s/le28r2/' % (locale,namespace) in r['note'],(locale,namespace,'live record not cited')
    peers=[e for e in new if (e['locale'],e['namespace'])==(locale,namespace)]
    assert peers[-1] is r,('refreshed row is not the pair latest',locale,namespace)
    assert sum(e['source_sha256']==r['source_sha256'] for e in peers)==1,('duplicate row at the current hash',locale,namespace)
    assert r['status'] in ('passed','blocked')
# This PR adds no ledger row at all: nothing the base did not hold may cite a bundle.
for i,e in enumerate(new):
    if i not in matched:
        assert 'audits/evidence/%s/%s/le28r2/' % (e['locale'],e['namespace']) not in e['note'],('appended recovery row',e['locale'],e['namespace'])
sha=lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
records=sorted((root/'i18n/audits/evidence').glob('*/*/le28r2/record.json'))
assert len(records)==len(rows)==24
files=0;inputs=0;counts={};routed=0;remediated=0;resolved_majors=0;majors_raised=0;majors_unresolved=0
for path in records:
    d=json.loads(path.read_text());b=path.parent
    r=json.loads((b/'receipt.json').read_text())
    assert set(r['hashes'])=={p.name for p in b.iterdir() if p.name!='receipt.json'}
    for name,h in r['hashes'].items():
        assert sha(b/name)==h,(str(b),name)
        files+=1
    for name,p in d['input_paths'].items():
        base_bytes=subprocess.check_output(['git','show',base+':'+p])
        assert hashlib.sha256(base_bytes).hexdigest()==d['input_hashes'][name],(str(b),name,'audited base bytes')
        inputs+=1
    row=rows[d['locale'],d['namespace']]
    assert row['source_sha256']==d['input_hashes']['en']
    assert row['status']==d['ledger_status'] in ('passed','blocked')
    route=json.loads((b/'route-receipts.json').read_text())
    assert [r['path'] for r in route['routed_reads']]==[route['routed_reads'][0]['path'], route['routed_reads'][1]['path']]
    assert len(route['routed_reads'])==2
    for rr in route['routed_reads']:
        assert rr['reader_body_sha256']==rr['recomputed_body_sha256'], rr['path']
        assert rr['tool']=='read_public_copy' and rr['key'] if False else rr['tool']=='read_public_copy'
        assert rr['session'] and rr['ts']
    assert route['context_sidecar']['refused'] is True
    assert d['read_provenance']['sanctioned_route_verified']=='partial'
    routed+=len(route['routed_reads'])
    coverage=json.loads((b/'coverage.json').read_text())
    assert len(coverage['keys'])==coverage['keys_read']==d['key_count']
    f=json.loads((b/'findings.json').read_text())
    assert f['unresolved_blocker_major']==sum(x['severity'] in ('blocker','major') for x in f['findings'])
    raised=sum(x['severity'] in ('blocker','major') for x in f['findings'])
    majors_raised+=raised
    rem=b.parent/'le28r2-remed'
    if rem.is_dir():
        rf=json.loads((rem/'findings.json').read_text())
        assert rf['verdict']=='PASS' and all(x['resolved'] for x in rf['findings'])
        rr=json.loads((rem/'route-receipt.json').read_text())
        blob=subprocess.check_output(['git','show',rr['recomputed_from_commit']+':'+rr['path']])
        assert hashlib.sha256(json.dumps(json.loads(blob),ensure_ascii=False,indent=1).encode()).hexdigest()==rr['sha256']
        # Post-review corrections (if any) must describe the bytes actually committed.
        pr=rf.get('post_review_delta')
        if pr:
            cat=root/d['input_paths']['target']
            assert pr['catalogue_sha256_after']==rf['target_sha256']==sha(cat),(str(rem),'post-review catalogue hash')
        remediated+=len(rf['findings'])
        resolved_majors+=sum(x['severity_raised'] in ('blocker','major') and x['resolved'] for x in rf['findings'])
    else:
        majors_unresolved+=raised
    counts[d['current_report_verdict']]=counts.get(d['current_report_verdict'],0)+1
# A hashed companion cannot be changed without detection (counterfactual local
# byte-only control, no mutation of tracked evidence).
p=records[0].parent/'report.md'
assert hashlib.sha256(p.read_bytes()+b'\nCORRUPTION-CONTROL\n').hexdigest()!=sha(p)
print(json.dumps({'result':'PASS','pairs':24,'original_rows_preserved':len(old),'ledger_entries_total':len(new),'historical_notes_appended':changed,'rows_refreshed_in_place':24,'rows_appended_by_this_pr':0,'companions_verified':files,'input_hashes_verified':inputs,'semantic_verdicts':counts,'majors_raised_before_remediation':majors_raised,'majors_resolved_by_remediation':resolved_majors,'unresolved_after_remediation':majors_unresolved,'remediated_findings_verified':remediated,'routed_file_receipts':routed,'pairs_with_context_refusal':24,'byte_corruption_negative_control':'detected','scope':'Integrity/history only; no sanctioned-reader or native/rendered certification.'},indent=2))
